// ===== 函数1: 快速检测图坐标 → 高分辨率原图坐标 =====
// 将 letterbox 后的检测结果坐标映射回 1920×1440 原图
cv::Rect map_lowres_to_highres(const object_detect_result &lowres_box, 
                                const letterbox_t &letterbox) {
    
    // 步骤1: 剔除letterbox填充 (640×640 → 480×640 旋转后坐标)
    float x_offset = letterbox.x_pad;  // 左右填充像素
    float y_offset = letterbox.y_pad;  // 上下填充像素
    float scale = letterbox.scale;     // letterbox缩放比例
    
    float x1_rot = (lowres_box.box.left - x_offset) / scale;
    float y1_rot = (lowres_box.box.top - y_offset) / scale;
    float x2_rot = (lowres_box.box.right - x_offset) / scale;
    float y2_rot = (lowres_box.box.bottom - y_offset) / scale;
    
    // 步骤2: 逆旋转 (480×640 旋转后 → 640×480 旋转前)
    // 逆时针90度旋转的逆变换: 
    // 已知旋转后坐标 (x', y')，旋转后尺寸 480×640
    // 旋转前坐标: x = height' - 1 - y', y = x'
    const int rotated_height = 640;
    
    float x1_orig = rotated_height - y2_rot;  // 注意用 y2
    float y1_orig = x1_rot;
    float x2_orig = rotated_height - y1_rot;  // 注意用 y1
    float y2_orig = x2_rot;
    
    // 步骤3: 缩放到高分辨率原图 (640×480 → 1920×1440)
    // 缩放比例均为 3.0
    const float scale_x = 1920.0f / 640.0f;  // = 3.0
    const float scale_y = 1440.0f / 480.0f;  // = 3.0
    
    int hx1 = (int)(x1_orig * scale_x);
    int hy1 = (int)(y1_orig * scale_y);
    int hx2 = (int)(x2_orig * scale_x);
    int hy2 = (int)(y2_orig * scale_y);
    
    // 步骤4: 边界裁剪
    hx1 = std::max(0, std::min(hx1, 1919));
    hy1 = std::max(0, std::min(hy1, 1439));
    hx2 = std::max(0, std::min(hx2, 1919));
    hy2 = std::max(0, std::min(hy2, 1439));
    
    return cv::Rect(hx1, hy1, hx2 - hx1, hy2 - hy1);
}

// ===== 函数2: 判断目标属于哪个9宫格分块 =====
// 输入: 高分辨率原图坐标
// 输出: 分块索引 (0-8) 和分块在原图中的ROI
struct BlockInfo {
    int block_id;         // 分块ID: 0-8
    cv::Rect block_roi;   // 分块在原图中的ROI (640×480)
    cv::Point offset;     // 分块左上角在原图中的偏移
};

BlockInfo get_block_info(const cv::Rect &highres_box) {
    // 9宫格布局:
    // 1920×1440 分为 3×3 块，每块 640×480
    // 
    //  0 | 1 | 2
    // ---|---|---
    //  3 | 4 | 5
    // ---|---|---
    //  6 | 7 | 8
    
    const int block_width = 640;
    const int block_height = 480;
    
    // 计算目标中心点
    int center_x = highres_box.x + highres_box.width / 2;
    int center_y = highres_box.y + highres_box.height / 2;
    
    // 确定所在列 (0, 1, 2)
    int col = std::min(center_x / block_width, 2);
    // 确定所在行 (0, 1, 2)
    int row = std::min(center_y / block_height, 2);
    
    BlockInfo info;
    info.block_id = row * 3 + col;  // 0-8
    
    // 计算分块的ROI
    info.offset.x = col * block_width;
    info.offset.y = row * block_height;
    info.block_roi = cv::Rect(info.offset.x, info.offset.y, 
                              block_width, block_height);
    
    return info;
}

// ===== 函数3: 分块检测结果 → 快速检测图坐标 (修正版) =====
// 将分块检测的结果映射回快速检测图坐标，用于绘制
cv::Rect map_block_to_lowres(const object_detect_result &block_box,
                              const letterbox_t &block_letterbox,
                              const BlockInfo &block_info) {
    
    // 步骤1: 剔除letterbox填充 (640×640 → 480×640 旋转后分块坐标)
    float x_offset = block_letterbox.x_pad;
    float y_offset = block_letterbox.y_pad;
    float scale = block_letterbox.scale;
    
    float bx1_rot = (block_box.box.left - x_offset) / scale;
    float by1_rot = (block_box.box.top - y_offset) / scale;
    float bx2_rot = (block_box.box.right - x_offset) / scale;
    float by2_rot = (block_box.box.bottom - y_offset) / scale;
    
    // 步骤2: 逆旋转分块坐标 (480×640 旋转后 → 640×480 旋转前)
    // 逆时针90度旋转的逆变换: x = height' - 1 - y', y = x'
    const int block_rotated_height = 640;
    
    float bx1 = block_rotated_height - by2_rot;  // 注意用 y2
    float by1 = bx1_rot;
    float bx2 = block_rotated_height - by1_rot;  // 注意用 y1
    float by2 = bx2_rot;
    
    // 步骤3: 加上分块在高分辨率原图中的偏移 (因为都在未旋转坐标系，可以直接相加)
    float hx1 = bx1 + block_info.offset.x;
    float hy1 = by1 + block_info.offset.y;
    float hx2 = bx2 + block_info.offset.x;
    float hy2 = by2 + block_info.offset.y;
    
    // 步骤4: 缩放回快速检测图尺寸 (1920×1440 → 640×480)
    // 缩放比例均为 1/3
    const float scale_x = 640.0f / 1920.0f;
    const float scale_y = 480.0f / 1440.0f;
    
    float lx1 = hx1 * scale_x;
    float ly1 = hy1 * scale_y;
    float lx2 = hx2 * scale_x;
    float ly2 = hy2 * scale_y;
    
    // 步骤5: 正向旋转到快速检测图坐标系 (640×480 → 480×640)
    // 逆时针90度旋转: x' = y, y' = width - x
    const int width_before_rotate = 640;
    
    float rx1 = ly1;
    float ry1 = width_before_rotate - lx2;  // 注意用 lx2
    float rx2 = ly2;
    float ry2 = width_before_rotate - lx1;  // 注意用 lx1
    
    // 确保坐标不越界（完整边界检查）
    rx1 = std::max(0.0f, std::min(rx1, 479.0f));
    ry1 = std::max(0.0f, std::min(ry1, 639.0f));
    rx2 = std::max(0.0f, std::min(rx2, 479.0f));
    ry2 = std::max(0.0f, std::min(ry2, 639.0f));
    
    // 确保宽高为正
    int width = std::max(1, (int)(rx2 - rx1));
    int height = std::max(1, (int)(ry2 - ry1));
    
    return cv::Rect((int)rx1, (int)ry1, width, height);
}

/*-------------------------------------------
                  Main Function
-------------------------------------------*/
int main(int argc, char **argv)
{
    // 检查命令行参数数量，需要传入模型路径
    if (argc != 2)
    {
        printf("%s <model_path>\n", argv[0]);
        return -1;
    }

    // 获取模型路径参数
    const char *model_path = argv[1];

    int ret;
    // 初始化RKNN应用上下文结构体
    rknn_app_context_t rknn_app_ctx;
    memset(&rknn_app_ctx, 0, sizeof(rknn_app_context_t));

    // 初始化后处理模块
    init_post_process();

    // 加载并初始化YOLOv8模型
    ret = init_yolov8_model(model_path, &rknn_app_ctx);
    if (ret != 0)
    {
        printf("init_yolov8_model fail! ret=%d model_path=%s\n", ret, model_path);
    }

    // 初始化源图像缓冲区和检测结果结构体
    image_buffer_t src_image;
    memset(&src_image, 0, sizeof(image_buffer_t));
    object_detect_result_list od_results;

    // ===== 任务 B.1：打开高分辨率摄像头 =====
    // 打开摄像头设备 (设备编号22)
    cv::VideoCapture cap(22);
    // ===== 修改：设置摄像头为 1920×1440 (4:3 比例) =====
    cap.set(cv::CAP_PROP_FRAME_WIDTH, 1920);
    cap.set(cv::CAP_PROP_FRAME_HEIGHT, 1440);
    
    if (!cap.isOpened()) {
        std::cerr << "Failed to open the camera." << std::endl;
        return -1;
    }
    
    // 验证实际分辨率
    int actual_width = cap.get(cv::CAP_PROP_FRAME_WIDTH);
    int actual_height = cap.get(cv::CAP_PROP_FRAME_HEIGHT);
    printf("Camera opened: %dx%d\n", actual_width, actual_height);

    // 创建名为"out"的OpenCV窗口
    cv::namedWindow("out", cv::WINDOW_NORMAL);
    // 设置窗口为全屏模式
    cv::setWindowProperty("out", cv::WND_PROP_FULLSCREEN, cv::WINDOW_FULLSCREEN);
    
    // ===== FPS 统计变量 =====
    int frame_count = 0;
    double total_time = 0.0;
    double preprocess_time = 0.0;
    double inference_time = 0.0;
    double postprocess_time = 0.0;

    // 预分配目标图像缓冲区，避免每帧 malloc/free
    image_buffer_t dst_img;
    memset(&dst_img, 0, sizeof(image_buffer_t));
    dst_img.width = rknn_app_ctx.model_width;
    dst_img.height = rknn_app_ctx.model_height;
    dst_img.format = IMAGE_FORMAT_RGB888;
    dst_img.size = get_image_size(&dst_img);

    // ===== 分配模型输入缓冲区 (使用DMA heap) =====
    // DMA heap是Linux提供的DMA内存分配机制
    // 可以分配物理连续的内存,供RGA等硬件加速器直接访问
    
    // 步骤1: 打开DMA heap设备文件
    // "/dev/dma_heap/system" 是系统默认的DMA heap设备
    int dma_heap_fd = open("/dev/dma_heap/system", O_RDWR);
    if (dma_heap_fd < 0) {
        // DMA heap打开失败 (可能是内核不支持或权限不足)
        printf("Failed to open dma_heap, fallback to malloc\n");
        // 降级方案: 使用普通malloc分配内存
        dst_img.virt_addr = (unsigned char*)malloc(dst_img.size);
        dst_img.fd = 0;  // fd=0 表示不是DMA缓冲区
    } else {
        // 步骤2: 准备DMA heap分配请求结构体
        struct dma_heap_allocation_data heap_data;
        memset(&heap_data, 0, sizeof(heap_data));
        heap_data.len = dst_img.size;           // 要分配的字节数 (640*640*3=1228800)
        heap_data.fd_flags = O_RDWR | O_CLOEXEC; // 文件描述符标志:
                                                   // O_RDWR: 可读写
                                                   // O_CLOEXEC: exec时自动关闭
        
        // 步骤3: 通过ioctl系统调用向DMA heap请求分配内存
        // DMA_HEAP_IOCTL_ALLOC 是DMA heap的分配命令
        if (ioctl(dma_heap_fd, DMA_HEAP_IOCTL_ALLOC, &heap_data) < 0) {
            // DMA分配失败 (可能是内存不足)
            printf("DMA heap alloc failed, fallback to malloc\n");
            dst_img.virt_addr = (unsigned char*)malloc(dst_img.size);
            dst_img.fd = 0;
        } else {
            // 步骤4: DMA分配成功,获取文件描述符
            // heap_data.fd 是内核返回的DMA缓冲区文件描述符
            // 这个fd代表一块物理连续的内存,可以传递给RGA等硬件
            dst_img.fd = heap_data.fd;
            
            // 步骤5: 将DMA缓冲区映射到用户空间虚拟地址
            // mmap系统调用建立虚拟地址到物理内存的映射关系
            dst_img.virt_addr = (unsigned char*)mmap(
                NULL,                    // 让内核自动选择虚拟地址
                dst_img.size,            // 映射大小
                PROT_READ | PROT_WRITE,  // 内存保护标志: 可读写
                MAP_SHARED,              // 映射类型: 共享映射 (硬件可见修改)
                dst_img.fd,              // DMA缓冲区的文件描述符
                0                        // 从文件开头偏移0字节开始映射
            );
            
            if (dst_img.virt_addr == MAP_FAILED) {
                // mmap失败 (通常不会发生,除非系统资源耗尽)
                printf("mmap failed, fallback to malloc\n");
                close(dst_img.fd);  // 关闭DMA缓冲区fd
                dst_img.virt_addr = (unsigned char*)malloc(dst_img.size);
                dst_img.fd = 0;
            } else {
                // 成功! 打印DMA缓冲区信息
                // 此时 dst_img.fd > 0, convert_image_rga会使用importbuffer_fd
                // RGA硬件可以通过fd直接访问物理内存,实现硬件加速
                printf("DMA buffer allocated: fd=%d size=%d\n", dst_img.fd, dst_img.size);
            }
        }
        
        // 步骤6: 关闭DMA heap设备文件描述符
        // 注意: 这里关闭的是 /dev/dma_heap/system 的fd,
        // 不是DMA缓冲区的fd (dst_img.fd仍然有效)
        close(dma_heap_fd);
    }
    
    // 最终检查: 确保缓冲区地址有效 (无论是DMA还是malloc)
    if (dst_img.virt_addr == NULL) {
        printf("malloc buffer size:%d fail!\n", dst_img.size);
        return -1;
    }

    // ===== 任务 B.1：高分辨率原图缓冲区 =====
    cv::Mat original_highres;           // 1920×1440 原图 (修改注释)
    cv::Mat shared_frame;               // 捕获线程共享缓冲区（保留原图）
    cv::Mat shared_detect_frame;        // 捕获线程共享的预缩放检测图
    std::mutex frame_mutex;             
    std::atomic<bool> stop_capture(false);

    // 捕获线程：持续读取高分辨率摄像头并同时生成检测用的缩放图，减少主线程开销
    std::thread cap_thread([&](){
        cv::Mat frame_bgr;
        cv::Mat frame_rgb;
        cv::Mat small;
        while(!stop_capture.load()) {
            if (!cap.read(frame_bgr)) continue;

            // ===== 重要：OpenCV 读取的是 BGR 格式，需要转换为 RGB =====
            cv::cvtColor(frame_bgr, frame_rgb, cv::COLOR_BGR2RGB);

            // 1920×1440 → 640×480 (保持4:3比例)
            cv::resize(frame_rgb, small, cv::Size(640, 480), 0, 0, cv::INTER_LINEAR);
            
            // 旋转90度到检测方向 (640×480 → 480×640)
            cv::rotate(small, small, cv::ROTATE_90_COUNTERCLOCKWISE);

            {
                std::lock_guard<std::mutex> lk(frame_mutex);
                frame_rgb.copyTo(shared_frame);
                small.copyTo(shared_detect_frame);
            }
        }
    });

    // 主循环：持续读取摄像头帧并进行目标检测
    while(true){
        // ===== 获取高分辨率原图及检测图 =====
        cv::Mat detect_frame_480p;
        {
            std::lock_guard<std::mutex> lk(frame_mutex);
            if (shared_detect_frame.empty()) {
                std::this_thread::sleep_for(std::chrono::milliseconds(1));
                continue;
            }
            shared_frame.copyTo(original_highres);
            shared_detect_frame.copyTo(detect_frame_480p);
        }

        auto frame_start = std::chrono::high_resolution_clock::now();
        auto preprocess_start = std::chrono::high_resolution_clock::now();

        // ===== 快速检测图预处理 =====
        src_image.height = detect_frame_480p.rows;      
        src_image.width = detect_frame_480p.cols;       
        src_image.width_stride = detect_frame_480p.step[0];
        src_image.virt_addr = detect_frame_480p.data;
        src_image.format = IMAGE_FORMAT_RGB888;
        src_image.size = detect_frame_480p.total() * detect_frame_480p.elemSize();

        letterbox_t letter_box;
        rknn_input inputs[rknn_app_ctx.io_num.n_input];
        rknn_output outputs[rknn_app_ctx.io_num.n_output];
        const float nms_threshold = NMS_THRESH;
        const float box_conf_threshold = BOX_THRESH;
        int bg_color = 0;

        memset(&od_results, 0x00, sizeof(od_results));
        memset(&letter_box, 0, sizeof(letterbox_t));
        
        ret = convert_image_with_letterbox(&src_image, &dst_img, &letter_box, bg_color);
        if (ret < 0) {
            printf("convert_image_with_letterbox fail! ret=%d\n", ret);
            return -1;
        }
        
        // ===== 快速检测推理 =====
        inputs[0].index = 0;
        inputs[0].type = RKNN_TENSOR_UINT8;
        inputs[0].fmt = RKNN_TENSOR_NHWC;
        inputs[0].size = rknn_app_ctx.model_width * rknn_app_ctx.model_height * rknn_app_ctx.model_channel;
        inputs[0].buf = dst_img.virt_addr;

        ret = rknn_inputs_set(rknn_app_ctx.rknn_ctx, rknn_app_ctx.io_num.n_input, inputs);
        if (ret < 0) {
            printf("rknn_input_set fail! ret=%d\n", ret);
            return -1;
        }
        
        auto preprocess_end = std::chrono::high_resolution_clock::now();
        double preprocess_ms = std::chrono::duration<double, std::milli>(preprocess_end - preprocess_start).count();
        preprocess_time += preprocess_ms;
        
        auto inference_start = std::chrono::high_resolution_clock::now();
        
        ret = rknn_run(rknn_app_ctx.rknn_ctx, nullptr);
        if (ret < 0) {
            printf("rknn_run fail! ret=%d\n", ret);
            return -1;
        }

        memset(outputs, 0, sizeof(outputs));
        for (int i = 0; i < rknn_app_ctx.io_num.n_output; i++) {
            outputs[i].index = i;
            outputs[i].want_float = (!rknn_app_ctx.is_quant);
        }
        ret = rknn_outputs_get(rknn_app_ctx.rknn_ctx, rknn_app_ctx.io_num.n_output, outputs, NULL);
        
        auto inference_end = std::chrono::high_resolution_clock::now();
        double inference_ms = std::chrono::duration<double, std::milli>(inference_end - inference_start).count();
        inference_time += inference_ms;

        auto postprocess_start = std::chrono::high_resolution_clock::now();
        
        post_process(&rknn_app_ctx, outputs, &letter_box, box_conf_threshold, nms_threshold, &od_results);
        rknn_outputs_release(rknn_app_ctx.rknn_ctx, rknn_app_ctx.io_num.n_output, outputs);

        // ===== 步骤1: 判断是否包含小目标并确定所在分块 =====
        bool has_small_target = false;
        std::set<int> target_blocks;  // 需要检测的分块ID集合
        
        for (int i = 0; i < od_results.count; i++) {
            object_detect_result *det_result = &(od_results.results[i]);
            int w = det_result->box.right - det_result->box.left;
            int h = det_result->box.bottom - det_result->box.top;
            
            // 判断是否为小目标（调整阈值：480×640图像上30像素更合理）
            if ((w < 30 || h < 30) && det_result->prop < 0.7) {
                has_small_target = true;
                
                // 映射到高分辨率原图坐标
                cv::Rect highres_box = map_lowres_to_highres(*det_result, letter_box);
                
                // 确定所在分块
                BlockInfo block_info = get_block_info(highres_box);
                target_blocks.insert(block_info.block_id);
                
                printf("[INFO] Small target: %s (%.1f%%), size=%dx%d, block=%d\n",
                       coco_cls_to_name(det_result->cls_id), 
                       det_result->prop * 100, w, h, block_info.block_id);
            }
        }

        // ===== 步骤2: 对包含小目标的分块进行高分辨率检测 =====
        object_detect_result_list highres_results;
        memset(&highres_results, 0x00, sizeof(highres_results));
        
        if (has_small_target && !original_highres.empty()) {
            printf("[INFO] Triggering high-res detection for %zu blocks...\n", target_blocks.size());
            auto highres_start = std::chrono::high_resolution_clock::now();
            
            for (int block_id : target_blocks) {
                // 计算分块的ROI
                int col = block_id % 3;
                int row = block_id / 3;
                cv::Rect block_roi(col * 640, row * 480, 640, 480);
                
                // 提取分块
                cv::Mat block_img = original_highres(block_roi);
                
                // 旋转分块
                cv::Mat block_rotated;
                cv::rotate(block_img, block_rotated, cv::ROTATE_90_COUNTERCLOCKWISE);
                
                // 准备分块输入
                image_buffer_t block_src;
                block_src.height = block_rotated.rows;  // 640
                block_src.width = block_rotated.cols;   // 480
                block_src.width_stride = block_rotated.step[0];
                block_src.virt_addr = block_rotated.data;
                block_src.format = IMAGE_FORMAT_RGB888;
                block_src.size = block_rotated.total() * block_rotated.elemSize();
                
                // Letterbox处理
                letterbox_t block_letterbox;
                memset(&block_letterbox, 0, sizeof(letterbox_t));
                
                ret = convert_image_with_letterbox(&block_src, &dst_img, &block_letterbox, bg_color);
                if (ret < 0) {
                    printf("Block %d convert failed! ret=%d\n", block_id, ret);
                    continue;
                }
                
                // 推理
                inputs[0].buf = dst_img.virt_addr;
                rknn_inputs_set(rknn_app_ctx.rknn_ctx, rknn_app_ctx.io_num.n_input, inputs);
                rknn_run(rknn_app_ctx.rknn_ctx, nullptr);
                
                memset(outputs, 0, sizeof(outputs));
                for (int i = 0; i < rknn_app_ctx.io_num.n_output; i++) {
                    outputs[i].index = i;
                    outputs[i].want_float = (!rknn_app_ctx.is_quant);
                }
                rknn_outputs_get(rknn_app_ctx.rknn_ctx, rknn_app_ctx.io_num.n_output, outputs, NULL);
                
                // 后处理分块结果
                object_detect_result_list block_results;
                memset(&block_results, 0x00, sizeof(block_results));
                post_process(&rknn_app_ctx, outputs, &block_letterbox, 
                            box_conf_threshold * 0.8, nms_threshold, &block_results);
                rknn_outputs_release(rknn_app_ctx.rknn_ctx, rknn_app_ctx.io_num.n_output, outputs);
                
                // 合并结果（映射坐标）
                BlockInfo block_info;
                block_info.block_id = block_id;
                block_info.offset = cv::Point(col * 640, row * 480);
                block_info.block_roi = block_roi;
                
                for (int i = 0; i < block_results.count && highres_results.count < OBJ_NUMB_MAX_SIZE; i++) {
                    // 将分块检测结果映射回快速检测图坐标
                    cv::Rect mapped_box = map_block_to_lowres(block_results.results[i], 
                                                              block_letterbox, block_info);
                    
                    // 保存映射后的结果
                    object_detect_result *hr_result = &highres_results.results[highres_results.count++];
                    *hr_result = block_results.results[i];
                    hr_result->box.left = mapped_box.x;
                    hr_result->box.top = mapped_box.y;
                    hr_result->box.right = mapped_box.x + mapped_box.width;
                    hr_result->box.bottom = mapped_box.y + mapped_box.height;
                }
                
                printf("[INFO] Block %d detected %d objects\n", block_id, block_results.count);
            }
            
            auto highres_end = std::chrono::high_resolution_clock::now();
            double highres_ms = std::chrono::duration<double, std::milli>(highres_end - highres_start).count();
            printf("[INFO] High-res detection took %.2f ms\n", highres_ms);
        }

        // ===== 在检测图上绘制结果 =====
        // 注意：detect_frame_480p 是 RGB 格式，直接在其上绘制
        // 绘制完成后再转换为 BGR 用于显示（避免不必要的 clone）
        cv::Mat result_mat_bgr;

        // 绘制快速检测结果（蓝色框）
        for (int i = 0; i < od_results.count; i++) {
            object_detect_result *det_result = &(od_results.results[i]);
            int x1 = det_result->box.left;
            int y1 = det_result->box.top;
            int x2 = det_result->box.right;
            int y2 = det_result->box.bottom;

            // RGB 格式下，蓝色是 (0, 0, 255)
            cv::rectangle(detect_frame_480p, cv::Point(x1, y1), cv::Point(x2, y2), 
                         cv::Scalar(0, 0, 255), 3);
            
            char text[256];
            sprintf(text, "%s %.1f%%", coco_cls_to_name(det_result->cls_id), det_result->prop * 100);
            // RGB 格式下，绿色是 (0, 255, 0)
            cv::putText(detect_frame_480p, text, cv::Point(x1, y1 - 10),
                       cv::FONT_HERSHEY_SIMPLEX, 0.5, cv::Scalar(0, 255, 0), 2);
        }
        
        // 绘制高分辨率检测结果（红色框）
        for (int i = 0; i < highres_results.count; i++) {
            object_detect_result *det_result = &(highres_results.results[i]);
            int x1 = det_result->box.left;
            int y1 = det_result->box.top;
            int x2 = det_result->box.right;
            int y2 = det_result->box.bottom;

            // RGB 格式下，红色是 (255, 0, 0)
            cv::rectangle(detect_frame_480p, cv::Point(x1, y1), cv::Point(x2, y2), 
                         cv::Scalar(255, 0, 0), 2);
            
            char text[256];
            sprintf(text, "HR:%s %.1f%%", coco_cls_to_name(det_result->cls_id), 
                   det_result->prop * 100);
            cv::putText(detect_frame_480p, text, cv::Point(x1, y1 - 30),
                       cv::FONT_HERSHEY_SIMPLEX, 0.4, cv::Scalar(255, 0, 0), 2);
        }

        // ===== 重要：OpenCV 的 imshow 期望 BGR 格式，需要转换 =====
        cv::cvtColor(detect_frame_480p, result_mat_bgr, cv::COLOR_RGB2BGR);
        cv::imshow("out", result_mat_bgr);

        int key = cv::waitKey(1);
        if (key == 'q' || key == 27) {
            break;
        }

        auto postprocess_end = std::chrono::high_resolution_clock::now();
        double postprocess_ms = std::chrono::duration<double, std::milli>(postprocess_end - postprocess_start).count();
        postprocess_time += postprocess_ms;

        auto frame_end = std::chrono::high_resolution_clock::now();
        double frame_ms = std::chrono::duration<double, std::milli>(frame_end - frame_start).count();
        total_time += frame_ms;
        frame_count++;
        
        if(frame_count%5==0) {
            double avg_fps = frame_count / (total_time/1000);
            double avg_total = (total_time) / frame_count;
            double avg_preprocess = (preprocess_time) / frame_count;
            double avg_inference = (inference_time) / frame_count;
            double avg_postprocess = (postprocess_time) / frame_count;

            printf("\n========== Performance Statistics ==========\n");
            printf("Camera Resolution: 1920x1440 (4:3)\n");
            printf("Detection Resolution: 640x480 → 640x640\n");
            printf("FPS: %.2f\n", avg_fps);
            printf("Average Total Time: %.2f ms\n", avg_total);
            printf("  - Preprocess:  %.2f ms (%.1f%%)\n", avg_preprocess, avg_preprocess/avg_total*100);
            printf("  - Inference:   %.2f ms (%.1f%%)\n", avg_inference, avg_inference/avg_total*100);
            printf("  - Postprocess: %.2f ms (%.1f%%)\n", avg_postprocess, avg_postprocess/avg_total*100);
            printf("===========================================\n\n");
        }
    }
    
    // ===== 清理资源 =====
    // 释放后处理模块资源
    deinit_post_process();

    // 释放YOLOv8模型资源
    ret = release_yolov8_model(&rknn_app_ctx);
    if (ret != 0)
    {
        printf("release_yolov8_model fail! ret=%d\n", ret);
    }

    // 释放源图像缓冲区（如果使用了malloc）
    if (src_image.virt_addr != NULL)
    {
        free(src_image.virt_addr);
    }

    // 释放目标图像缓冲区
    if (dst_img.fd > 0) {
        // 如果使用DMA缓冲区，先取消映射再关闭fd
        munmap(dst_img.virt_addr, dst_img.size);
        close(dst_img.fd);
    } else if (dst_img.virt_addr) {
        // 如果使用malloc，直接释放
        free(dst_img.virt_addr);
    }

    // 停止捕获线程
    stop_capture.store(true);
    if (cap_thread.joinable()) cap_thread.join();

    return 0;
}
